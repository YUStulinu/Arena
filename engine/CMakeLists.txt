cmake_minimum_required(VERSION 3.20)
project(arena_engine C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
if(MINGW)
  add_link_options(-static)
endif()

# Lua 5.4 (vendored, MIT), without the stand-alone interpreter and compiler
file(GLOB LUA_SRC ../third_party/lua/src/*.c)
list(REMOVE_ITEM LUA_SRC ${CMAKE_CURRENT_SOURCE_DIR}/../third_party/lua/src/lua.c ${CMAKE_CURRENT_SOURCE_DIR}/../third_party/lua/src/luac.c)
add_library(lua STATIC ${LUA_SRC})
target_include_directories(lua PUBLIC ../third_party/lua/src)
set_target_properties(lua PROPERTIES POSITION_INDEPENDENT_CODE ON)

add_library(arena STATIC src/fixed.cpp src/game.cpp src/session.cpp src/capi.cpp)
target_include_directories(arena PUBLIC include)
target_link_libraries(arena PUBLIC lua)
set_target_properties(arena PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(arena PRIVATE -Wall -Wextra)
endif()

# a shared library for the Python bots (ctypes)
add_library(arena_shared SHARED src/fixed.cpp src/game.cpp src/session.cpp src/capi.cpp)
target_include_directories(arena_shared PUBLIC include)
target_compile_definitions(arena_shared PRIVATE ARENA_SHARED)
target_link_libraries(arena_shared PRIVATE lua)
set_target_properties(arena_shared PROPERTIES OUTPUT_NAME arena PREFIX "lib")

enable_testing()
add_executable(test_engine tests/test_engine.cpp)
target_link_libraries(test_engine PRIVATE arena)
add_test(NAME test_engine COMMAND test_engine WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR})
